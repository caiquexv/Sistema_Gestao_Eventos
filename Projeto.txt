1. DESCRICAO FINAL DO SISTEMA
============================================================
O sistema implementa a gestao de eventos e dos espacos onde eles acontecem. Cada evento possui um periodo de realizacao, pode pertencer a uma serie que agrupa suas edicoes recorrentes e pode ser promovido por pessoas ou organizacoes. A programacao de um evento e formada por atividades, e cada atividade pode ocorrer em uma ou mais sessoes, sendo cada sessao realizada em uma data, um horario e um espaco. Os espacos possuem capacidade e um conjunto de recursos. As pessoas se relacionam com os eventos como promotoras, participantes inscritas ou organizadoras, e com as atividades como palestrantes, instrutoras ou mediadoras. As sessoes restritas exigem inscricao adicional e respeitam um limite de vagas. O banco garante a coerencia temporal das sessoes, impede conflito de espaco, controla a admissao em sessoes restritas e preserva o historico de alteracoes de programacao por meio de triggers.
============================================================
2. DECISOES DE PROJETO
============================================================
Q1: Pessoas e organizacoes sao entidades distintas e um evento pode ter varios promotores, representados por duas associacoes de promocao, uma para pessoas e outra para organizacoes, o que permite tanto promocao conjunta quanto de naturezas mistas.
Q2: Cada pessoa possui cadastro unico, a participacao como ouvinte corresponde a inscricao no evento e os papeis de atuacao ficam registrados a parte, com palestrante e instrutor ligados a atividade e organizador ligado ao evento, podendo uma mesma pessoa acumular papeis.
Q3: A atividade guarda titulo, descricao e tipo e pertence a um evento, enquanto cada sessao guarda a data, o horario e o espaco de uma realizacao concreta, de modo que uma atividade pode ter varias sessoes sem duplicar informacao.
Q4: Cada edicao e modelada como um evento proprio com periodo e programacao especificos, e eventos recorrentes sao agrupados por uma serie opcional que guarda a identidade compartilhada, permitindo tambem eventos avulsos sem serie.
Q5: A inscricao especifica e exigida apenas nas sessoes marcadas como restritas e pressupoe inscricao confirmada no evento, sendo o limite efetivo da sessao o seu limite de vagas quando informado ou a capacidade do espaco caso contrario, e inscricoes que ultrapassariam esse limite sao recusadas.
Q6: A programacao vigente permanece na sessao e toda alteracao de data, horario ou espaco e registrada em uma tabela de historico com os valores anteriores e novos e o instante da mudanca, por meio de trigger, preservando o historico.

-- ---------- Cadastros base ----------
CREATE TABLE serie (
    id    bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome  varchar(150) NOT NULL UNIQUE
);

CREATE TABLE organizacao (
    id    bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome  varchar(150) NOT NULL,
    tipo  varchar(50)
);

CREATE TABLE pessoa (
    id     bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome   varchar(120) NOT NULL,
    email  varchar(160) NOT NULL UNIQUE
);

CREATE TABLE espaco (
    id          bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome        varchar(100) NOT NULL UNIQUE,
    capacidade  integer NOT NULL CHECK (capacidade > 0)
);

CREATE TABLE recurso (
    id    bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nome  varchar(60) NOT NULL UNIQUE
);

-- ---------- Estrutura do evento ----------
CREATE TABLE evento (
    id           bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    serie_id     bigint REFERENCES serie(id) ON DELETE SET NULL ON UPDATE CASCADE,
    nome         varchar(150) NOT NULL,
    data_inicio  date NOT NULL,
    data_fim     date NOT NULL,
    CONSTRAINT periodo_valido CHECK (data_fim >= data_inicio)
);

CREATE TABLE atividade (
    id         bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    evento_id  bigint NOT NULL REFERENCES evento(id) ON DELETE CASCADE ON UPDATE CASCADE,
    titulo     varchar(150) NOT NULL,
    descricao  text,
    tipo       varchar(40) NOT NULL
               CHECK (tipo IN ('palestra','workshop','minicurso','painel','mesa-redonda','apresentacao'))
);

CREATE TABLE sessao (
    id                bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    atividade_id      bigint NOT NULL REFERENCES atividade(id) ON DELETE CASCADE ON UPDATE CASCADE,
    espaco_id         bigint NOT NULL REFERENCES espaco(id) ON DELETE RESTRICT ON UPDATE CASCADE,
    data              date NOT NULL,
    hora_inicio       time NOT NULL,
    hora_fim          time NOT NULL,
    duracao           interval GENERATED ALWAYS AS (hora_fim - hora_inicio) STORED,
    requer_inscricao  boolean NOT NULL DEFAULT false,
    limite_vagas      integer,
    CONSTRAINT horario_valido CHECK (hora_fim > hora_inicio),
    CONSTRAINT limite_vagas_positivo CHECK (limite_vagas IS NULL OR limite_vagas > 0)
);

-- ---------- Promocao e recursos ----------
CREATE TABLE promocao_pessoa (
    evento_id  bigint NOT NULL REFERENCES evento(id) ON DELETE CASCADE ON UPDATE CASCADE,
    pessoa_id  bigint NOT NULL REFERENCES pessoa(id) ON DELETE CASCADE ON UPDATE CASCADE,
    PRIMARY KEY (evento_id, pessoa_id)
);

CREATE TABLE promocao_organizacao (
    evento_id       bigint NOT NULL REFERENCES evento(id) ON DELETE CASCADE ON UPDATE CASCADE,
    organizacao_id  bigint NOT NULL REFERENCES organizacao(id) ON DELETE CASCADE ON UPDATE CASCADE,
    PRIMARY KEY (evento_id, organizacao_id)
);

CREATE TABLE espaco_recurso (
    espaco_id   bigint NOT NULL REFERENCES espaco(id) ON DELETE CASCADE ON UPDATE CASCADE,
    recurso_id  bigint NOT NULL REFERENCES recurso(id) ON DELETE CASCADE ON UPDATE CASCADE,
    PRIMARY KEY (espaco_id, recurso_id)
);

-- ---------- Papeis das pessoas ----------
CREATE TABLE papel_evento (
    pessoa_id  bigint NOT NULL REFERENCES pessoa(id) ON DELETE CASCADE ON UPDATE CASCADE,
    evento_id  bigint NOT NULL REFERENCES evento(id) ON DELETE CASCADE ON UPDATE CASCADE,
    papel      varchar(30) NOT NULL DEFAULT 'organizador'
               CHECK (papel IN ('organizador','coordenador')),
    PRIMARY KEY (pessoa_id, evento_id, papel)
);

CREATE TABLE papel_atividade (
    pessoa_id     bigint NOT NULL REFERENCES pessoa(id) ON DELETE CASCADE ON UPDATE CASCADE,
    atividade_id  bigint NOT NULL REFERENCES atividade(id) ON DELETE CASCADE ON UPDATE CASCADE,
    papel         varchar(30) NOT NULL
                  CHECK (papel IN ('palestrante','instrutor','mediador')),
    PRIMARY KEY (pessoa_id, atividade_id, papel)
);

-- ---------- Inscricoes ----------
CREATE TABLE inscricao_evento (
    pessoa_id       bigint NOT NULL REFERENCES pessoa(id) ON DELETE CASCADE ON UPDATE CASCADE,
    evento_id       bigint NOT NULL REFERENCES evento(id) ON DELETE CASCADE ON UPDATE CASCADE,
    status          varchar(20) NOT NULL DEFAULT 'confirmada'
                    CHECK (status IN ('confirmada','cancelada')),
    data_inscricao  timestamp NOT NULL DEFAULT now(),
    PRIMARY KEY (pessoa_id, evento_id)
);

CREATE TABLE inscricao_sessao (
    pessoa_id       bigint NOT NULL REFERENCES pessoa(id) ON DELETE CASCADE ON UPDATE CASCADE,
    sessao_id       bigint NOT NULL REFERENCES sessao(id) ON DELETE CASCADE ON UPDATE CASCADE,
    status          varchar(20) NOT NULL DEFAULT 'confirmada'
                    CHECK (status IN ('confirmada','cancelada')),
    data_inscricao  timestamp NOT NULL DEFAULT now(),
    PRIMARY KEY (pessoa_id, sessao_id)
);

-- ---------- Historico ----------
CREATE TABLE historico_sessao (
    id                    bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    sessao_id             bigint NOT NULL REFERENCES sessao(id) ON DELETE CASCADE ON UPDATE CASCADE,
    data_anterior         date NOT NULL,
    hora_inicio_anterior  time NOT NULL,
    hora_fim_anterior     time NOT NULL,
    espaco_anterior_id    bigint NOT NULL REFERENCES espaco(id) ON DELETE RESTRICT ON UPDATE CASCADE,
    data_nova             date NOT NULL,
    hora_inicio_nova      time NOT NULL,
    hora_fim_nova         time NOT NULL,
    espaco_novo_id        bigint NOT NULL REFERENCES espaco(id) ON DELETE RESTRICT ON UPDATE CASCADE,
    alterado_em           timestamp NOT NULL DEFAULT now()
);

-- ---------- Trigger 1: coerencia temporal da sessao ----------
CREATE OR REPLACE FUNCTION fn_sessao_coerencia_temporal()
RETURNS trigger AS $$
DECLARE
    v_inicio date;
    v_fim    date;
BEGIN
    SELECT e.data_inicio, e.data_fim
      INTO v_inicio, v_fim
      FROM atividade a
      JOIN evento e ON e.id = a.evento_id
     WHERE a.id = NEW.atividade_id;

    IF NEW.data < v_inicio OR NEW.data > v_fim THEN
        RAISE EXCEPTION 'A data da sessao (%) esta fora do periodo do evento (% a %).',
            NEW.data, v_inicio, v_fim;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_sessao_coerencia_temporal
    BEFORE INSERT OR UPDATE ON sessao
    FOR EACH ROW EXECUTE FUNCTION fn_sessao_coerencia_temporal();

-- ---------- Trigger 2: conflito de espaco ----------
CREATE OR REPLACE FUNCTION fn_sessao_conflito_espaco()
RETURNS trigger AS $$
DECLARE
    v_conflitos integer;
BEGIN
    SELECT count(*)
      INTO v_conflitos
      FROM sessao s
     WHERE s.espaco_id = NEW.espaco_id
       AND s.data = NEW.data
       AND s.id <> NEW.id
       AND NEW.hora_inicio < s.hora_fim
       AND NEW.hora_fim > s.hora_inicio;

    IF v_conflitos > 0 THEN
        RAISE EXCEPTION 'Conflito de espaco: o espaco % ja possui sessao sobreposta no dia %.',
            NEW.espaco_id, NEW.data;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_sessao_conflito_espaco
    BEFORE INSERT OR UPDATE ON sessao
    FOR EACH ROW EXECUTE FUNCTION fn_sessao_conflito_espaco();

-- ---------- Trigger 3: admissao em sessao (inscricao no evento + vagas) ----------
CREATE OR REPLACE FUNCTION fn_inscricao_sessao_admissao()
RETURNS trigger AS $$
DECLARE
    v_evento_id       bigint;
    v_limite_vagas    integer;
    v_capacidade      integer;
    v_limite_efetivo  integer;
    v_ocupacao        integer;
    v_inscrito_evento integer;
BEGIN
    IF NEW.status <> 'confirmada' THEN
        RETURN NEW;
    END IF;

    SELECT e.id, s.limite_vagas, ep.capacidade
      INTO v_evento_id, v_limite_vagas, v_capacidade
      FROM sessao s
      JOIN atividade a ON a.id = s.atividade_id
      JOIN evento e    ON e.id = a.evento_id
      JOIN espaco ep   ON ep.id = s.espaco_id
     WHERE s.id = NEW.sessao_id;

    SELECT count(*)
      INTO v_inscrito_evento
      FROM inscricao_evento ie
     WHERE ie.pessoa_id = NEW.pessoa_id
       AND ie.evento_id = v_evento_id
       AND ie.status = 'confirmada';

    IF v_inscrito_evento = 0 THEN
        RAISE EXCEPTION 'A pessoa % nao possui inscricao confirmada no evento % da sessao.',
            NEW.pessoa_id, v_evento_id;
    END IF;

    v_limite_efetivo := COALESCE(v_limite_vagas, v_capacidade);

    SELECT count(*)
      INTO v_ocupacao
      FROM inscricao_sessao i
     WHERE i.sessao_id = NEW.sessao_id
       AND i.status = 'confirmada'
       AND NOT (i.pessoa_id = NEW.pessoa_id AND i.sessao_id = NEW.sessao_id);

    IF v_ocupacao + 1 > v_limite_efetivo THEN
        RAISE EXCEPTION 'Sessao % lotada: limite de % participantes atingido.',
            NEW.sessao_id, v_limite_efetivo;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_inscricao_sessao_admissao
    BEFORE INSERT OR UPDATE ON inscricao_sessao
    FOR EACH ROW EXECUTE FUNCTION fn_inscricao_sessao_admissao();

-- ---------- Trigger 4: historico de programacao ----------
CREATE OR REPLACE FUNCTION fn_sessao_historico()
RETURNS trigger AS $$
BEGIN
    IF (OLD.data, OLD.hora_inicio, OLD.hora_fim, OLD.espaco_id)
       IS DISTINCT FROM
       (NEW.data, NEW.hora_inicio, NEW.hora_fim, NEW.espaco_id) THEN

        INSERT INTO historico_sessao (
            sessao_id,
            data_anterior, hora_inicio_anterior, hora_fim_anterior, espaco_anterior_id,
            data_nova, hora_inicio_nova, hora_fim_nova, espaco_novo_id
        ) VALUES (
            OLD.id,
            OLD.data, OLD.hora_inicio, OLD.hora_fim, OLD.espaco_id,
            NEW.data, NEW.hora_inicio, NEW.hora_fim, NEW.espaco_id
        );
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER tg_sessao_historico
    AFTER UPDATE ON sessao
    FOR EACH ROW EXECUTE FUNCTION fn_sessao_historico();

CONSULTA 1
OBJETIVO: Lista a programacao completa de um evento, mostrando cada atividade com a data, o horario e o espaco de suas sessoes, em ordem cronologica.
SQL:
SELECT e.nome AS evento, a.titulo AS atividade,
       s.data, s.hora_inicio, s.hora_fim, ep.nome AS espaco
FROM evento e
JOIN atividade a ON a.evento_id = e.id
JOIN sessao s ON s.atividade_id = a.id
JOIN espaco ep ON ep.id = s.espaco_id
WHERE e.id = 2
ORDER BY s.data, s.hora_inicio;

CONSULTA 2
OBJETIVO: Mostra, para cada sessao que exige inscricao, quantas inscricoes confirmadas ela possui, qual o limite efetivo de participantes e quantas vagas ainda restam, incluindo sessoes sem inscritos.
SQL:
SELECT s.id AS sessao_id, a.titulo AS atividade, s.data,
       COALESCE(s.limite_vagas, ep.capacidade) AS limite_efetivo,
       count(i.pessoa_id) FILTER (WHERE i.status = 'confirmada') AS confirmadas,
       COALESCE(s.limite_vagas, ep.capacidade)
           - count(i.pessoa_id) FILTER (WHERE i.status = 'confirmada') AS vagas_restantes
FROM sessao s
JOIN atividade a ON a.id = s.atividade_id
JOIN espaco ep ON ep.id = s.espaco_id
LEFT JOIN inscricao_sessao i ON i.sessao_id = s.id
WHERE s.requer_inscricao = true
GROUP BY s.id, a.titulo, s.data, s.limite_vagas, ep.capacidade
ORDER BY s.id;

CONSULTA 3
OBJETIVO: Retorna as pessoas que confirmaram inscricao no evento 2 e tambem confirmaram inscricao em todas as sessoes restritas desse evento.
SQL:
SELECT p.id, p.nome
FROM pessoa p
JOIN inscricao_evento ie
     ON ie.pessoa_id = p.id AND ie.evento_id = 2 AND ie.status = 'confirmada'
WHERE NOT EXISTS (
    SELECT 1
    FROM sessao s
    JOIN atividade a ON a.id = s.atividade_id
    WHERE a.evento_id = 2 AND s.requer_inscricao = true
      AND NOT EXISTS (
          SELECT 1 FROM inscricao_sessao i
          WHERE i.sessao_id = s.id
            AND i.pessoa_id = p.id
            AND i.status = 'confirmada'
      )
)
ORDER BY p.id;

CONSULTA 4
OBJETIVO: Apresenta, para cada evento, as tres sessoes com maior taxa de ocupacao, com a contagem de confirmados, a capacidade do espaco e a posicao no ranking.
SQL:
WITH ocupacao AS (
    SELECT e.id AS evento_id, s.id AS sessao_id, a.titulo AS atividade,
           ep.capacidade,
           count(i.pessoa_id) FILTER (WHERE i.status = 'confirmada') AS confirmadas
    FROM evento e
    JOIN atividade a ON a.evento_id = e.id
    JOIN sessao s ON s.atividade_id = a.id
    JOIN espaco ep ON ep.id = s.espaco_id
    LEFT JOIN inscricao_sessao i ON i.sessao_id = s.id
    GROUP BY e.id, s.id, a.titulo, ep.capacidade
),
ranking AS (
    SELECT o.*,
           round(o.confirmadas::numeric / o.capacidade, 3) AS taxa_ocupacao,
           row_number() OVER (
               PARTITION BY o.evento_id
               ORDER BY o.confirmadas::numeric / o.capacidade DESC, o.sessao_id
           ) AS posicao
    FROM ocupacao o
)
SELECT evento_id, sessao_id, atividade, confirmadas, capacidade, taxa_ocupacao, posicao
FROM ranking
WHERE posicao <= 3
ORDER BY evento_id, posicao;

CONSULTA 5
OBJETIVO: Indica os espacos com capacidade de pelo menos 50 lugares que estao livres em 15 de setembro de 2026 no intervalo entre 9h e 11h.
SQL:
SELECT ep.id, ep.nome, ep.capacidade
FROM espaco ep
WHERE ep.capacidade >= 50
  AND NOT EXISTS (
      SELECT 1 FROM sessao s
      WHERE s.espaco_id = ep.id
        AND s.data = DATE '2026-09-15'
        AND s.hora_inicio < TIME '11:00'
        AND s.hora_fim > TIME '09:00'
  )
ORDER BY ep.capacidade DESC;
